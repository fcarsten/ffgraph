cmake_minimum_required(VERSION 3.24)
project(ffgraph_external_consumer LANGUAGES CXX)
find_package(ffgraph 1.0 CONFIG REQUIRED)
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE ffgraph::ffgraph)
if(FFGRAPH_CONSUMER_SANITIZERS)
  target_compile_options(consumer PRIVATE -fsanitize=address,undefined,float-cast-overflow -fno-omit-frame-pointer)
  target_link_options(consumer PRIVATE -fsanitize=address,undefined,float-cast-overflow)
endif()
